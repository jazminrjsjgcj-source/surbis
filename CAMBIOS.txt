python3 - <<'PY'
import re

p = 'routes/web.php'
s = open(p).read()

if 'kiosk.link' in s:
    print("rutas: ya estaban")
else:
    nuevas = """

/*
 * El quiosco. SIN sesion de usuario: la tableta se identifica con su
 * credencial en cookie, que ResolveKioskStation comprueba en CADA peticion
 * —asi revocarla apaga la tableta en el momento siguiente—.
 */
Route::middleware(ResolveKioskStation::class)->group(function (): void {
    Route::get('/kiosk/vincular', [LinkController::class, 'show'])->name('kiosk.link');
    Route::post('/kiosk/vincular', [LinkController::class, 'store'])->name('kiosk.link.store');

    Route::get('/kiosk/preparar', [KioskController::class, 'prepare'])->name('kiosk.prepare');
    Route::post('/kiosk/preparar', [KioskController::class, 'open'])->name('kiosk.prepare.store');

    Route::get('/kiosk', [KioskController::class, 'welcome'])->name('kiosk.welcome');
    Route::post('/kiosk/responder', KioskResponseController::class)->name('kiosk.submit');
});
"""

    for imp in ["use App\\Http\\Controllers\\Kiosk\\KioskController;",
                "use App\\Http\\Controllers\\Kiosk\\KioskResponseController;",
                "use App\\Http\\Controllers\\Kiosk\\LinkController;",
                "use App\\Http\\Middleware\\ResolveKioskStation;"]:
        if imp not in s:
            s = s.replace("use App\\Http\\Controllers\\HomeController;",
                          "use App\\Http\\Controllers\\HomeController;\n" + imp, 1)

    m = list(re.finditer(r'^use\s+[\w\\]+;$', s, re.M))
    a, b = m[0].start(), m[-1].end()
    s = s[:a] + '\n'.join(sorted([x.group(0) for x in m], key=lambda l: l[4:].lower())) + s[b:]

    open(p, 'w').write(s.rstrip() + nuevas)
    print("rutas anadidas")
PY

## Lo que queda preparado para la Fase 10

El envio del quiosco tiene UN solo punto —la funcion `enviar` de Welcome.tsx—
y el UUID se genera al empezar CADA respuesta, no al cargar la pantalla. Una
tableta no se recarga en todo el dia: si el UUID fuera de la pantalla, la
segunda persona reutilizaria el de la primera y su respuesta se descartaria
por idempotente.

La cola local de la Fase 10 sustituye esa funcion sin tocar las pantallas.
