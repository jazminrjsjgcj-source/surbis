## 1. Ruta de exportacion y enlace

cd ~/encuestas
python3 - <<'PY'
import re

p = 'routes/web.php'
s = open(p).read()

if 'admin.analytics.export' in s:
    print("ruta: ya estaba")
else:
    ancla = "        Route::get('/admin/analisis', AnalyticsController::class)->name('admin.analytics');"
    nueva = ancla + """
        Route::get('/admin/analisis/exportar', AnalyticsExportController::class)
            ->name('admin.analytics.export');"""

    if ancla not in s:
        print("NO encontrado el ancla")
    else:
        s = s.replace(ancla, nueva, 1)

        if 'use App\\Http\\Controllers\\Admin\\AnalyticsExportController;' not in s:
            s = s.replace("use App\\Http\\Controllers\\Admin\\AnalyticsController;",
                          "use App\\Http\\Controllers\\Admin\\AnalyticsController;\nuse App\\Http\\Controllers\\Admin\\AnalyticsExportController;", 1)

        m = list(re.finditer(r'^use\s+[\w\\]+;$', s, re.M))
        a, b = m[0].start(), m[-1].end()
        s = s[:a] + '\n'.join(sorted([x.group(0) for x in m], key=lambda l: l[4:].lower())) + s[b:]
        open(p, 'w').write(s)
        print("ruta anadida")
PY
