import re

p = 'routes/web.php'
s = open(p).read()

if 'admin.analytics' in s:
    print("ruta: ya estaba")
else:
    ancla = "        Route::get('/admin/respuestas', [ResponseController::class, 'index'])\n            ->name('admin.responses.index');"
    nueva = ancla + """

        Route::get('/admin/analisis', AnalyticsController::class)->name('admin.analytics');"""

    if ancla not in s:
        print("NO encontrado el ancla")
    else:
        s = s.replace(ancla, nueva, 1)

        if 'use App\\Http\\Controllers\\Admin\\AnalyticsController;' not in s:
            s = s.replace("use App\\Http\\Controllers\\Admin\\AreaController;",
                          "use App\\Http\\Controllers\\Admin\\AnalyticsController;\nuse App\\Http\\Controllers\\Admin\\AreaController;", 1)

        m = list(re.finditer(r'^use\s+[\w\\]+;$', s, re.M))
        a, b = m[0].start(), m[-1].end()
        s = s[:a] + '\n'.join(sorted([x.group(0) for x in m], key=lambda l: l[4:].lower())) + s[b:]
        open(p, 'w').write(s)
        print("ruta anadida")

# El menu
p2 = 'app/Http/Middleware/HandleInertiaRequests.php'
t = open(p2).read()

if "'analytics'" in t:
    print("nav: ya estaba")
else:
    ancla2 = "            ['key' => 'responses', 'url' => route('admin.responses.index')],"
    t = t.replace(ancla2, ancla2 + "\n            ['key' => 'analytics', 'url' => route('admin.analytics')],", 1)
    open(p2, 'w').write(t)
    print("nav anadido")

# Los contadores del menu: de 6 a 7
for ruta in ('tests/Feature/Admin/AdminShellTest.php',
             'tests/Feature/Admin/DashboardTest.php'):
    u = open(ruta).read()
    n = u.count("->has('nav', 6)")
    if n:
        open(ruta, 'w').write(u.replace("->has('nav', 6)", "->has('nav', 7)"))
        print(f"{ruta.split('/')[-1]}: {n} contador(es) a 7")

## Lo que cierra esta fase

El umbral se aplica en QueryMetrics y NADA lo esquiva: tarjetas, serie por
dia, comparaciones por sucursal, area, persona y canal.

Y los dias sin datos suficientes se QUITAN del grafico en vez de dibujarse en
cero: un punto oculto y uno con valor cero se ven igual en una linea, y eso
haria leer una caida donde solo hay pocos datos.
