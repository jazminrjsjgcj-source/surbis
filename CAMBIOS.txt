import re

# Las columnas de la clave, que TASK-039 dejo fuera del modelo
p0 = 'app/Domain/Organizations/Models/Device.php'
s0 = open(p0).read()

if 'station_key_hash' in s0:
    print("modelo: ya estaba")
else:
    s0 = s0.replace("        'status',\n", """        'status',
        'station_key_hash',
        'station_key_set_at',
        'station_key_revoked_at',
""", 1)
    s0 = s0.replace("""    public function isActive(): bool""", """    /** @return array<string, string> */
    protected function casts(): array
    {
        return [
            'station_key_set_at' => 'datetime',
            'station_key_revoked_at' => 'datetime',
        ];
    }

    public function isActive(): bool""", 1)
    open(p0, 'w').write(s0)
    print("modelo actualizado")

# Rutas
p = 'routes/web.php'
s = open(p).read()

if 'admin.branches.kiosks' in s:
    print("rutas: ya estaban")
else:
    ancla = "            ->name('admin.branches.activate');"
    nuevas = ancla + """

        /*
         * Los quioscos de una sucursal. El interruptor es una operacion en
         * LOTE sobre los deployments de sus dispositivos, no un deployment
         * de sucursal.
         */
        Route::get('/admin/sucursales/{branch}/quioscos', [BranchKioskController::class, 'show'])
            ->name('admin.branches.kiosks');
        Route::post('/admin/sucursales/{branch}/quioscos/activar', [BranchKioskController::class, 'activate'])
            ->name('admin.branches.kiosks.activate');
        Route::post('/admin/sucursales/{branch}/quioscos/suspender', [BranchKioskController::class, 'suspend'])
            ->name('admin.branches.kiosks.suspend');

        Route::post('/admin/dispositivos/{device}/clave', [DeviceKeyController::class, 'generate'])
            ->name('admin.devices.key.generate');
        Route::post('/admin/dispositivos/{device}/clave/revocar', [DeviceKeyController::class, 'revoke'])
            ->name('admin.devices.key.revoke');"""

    if ancla not in s:
        print("NO encontrado el ancla de rutas")
    else:
        s = s.replace(ancla, nuevas, 1)
        for imp in ["use App\\Http\\Controllers\\Admin\\BranchKioskController;",
                    "use App\\Http\\Controllers\\Admin\\DeviceKeyController;"]:
            if imp not in s:
                s = s.replace("use App\\Http\\Controllers\\Admin\\BranchController;",
                              "use App\\Http\\Controllers\\Admin\\BranchController;\n" + imp, 1)
        m = list(re.finditer(r'^use\s+[\w\\]+;$', s, re.M))
        a, b = m[0].start(), m[-1].end()
        s = s[:a] + '\n'.join(sorted([x.group(0) for x in m], key=lambda l: l[4:].lower())) + s[b:]
        open(p, 'w').write(s)
        print("rutas anadidas")

# Policy
p2 = 'app/Providers/AppServiceProvider.php'
t = open(p2).read()

if 'DevicePolicy' in t:
    print("policy: ya estaba")
else:
    ancla2 = "        Gate::policy(Area::class, AreaPolicy::class);"

    for imp in ["use App\\Domain\\Organizations\\Models\\Device;", "use App\\Policies\\DevicePolicy;"]:
        if imp not in t:
            t = t.replace("use App\\Policies\\AreaPolicy;", imp + "\nuse App\\Policies\\AreaPolicy;", 1)
    m = list(re.finditer(r'^use\s+[\w\\]+;$', t, re.M))
    a, b = m[0].start(), m[-1].end()
    t = t[:a] + '\n'.join(sorted([x.group(0) for x in m], key=lambda l: l[4:].lower())) + t[b:]
    open(p2, 'w').write(t)
    print("policy registrada")
PY

OJO: la clave 'key_revoked' se usa dos veces con sentidos distintos —el aviso
y el estado—. Si TranslationKeysTest se queja, renombra el estado a
'key_revoked_state' en Kiosks.tsx.

## 3. Un enlace desde la sucursal

En Admin/Branches/Form.tsx, junto a los otros botones:

    <Link href={kiosksUrl} className="btn btn-ghost">
        {t('interface.kiosks.title')}
    </Link>

Con 'kiosksUrl' => route('admin.branches.kiosks', $branch) en
BranchController::edit().
